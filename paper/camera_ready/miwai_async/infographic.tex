% MIWAI 2026, paper 122. One-page infographic for the asynchronous session.
%
% Canvas 320 x 180 mm (16:9), exported to 1920 x 1080, so one aspect ratio
% serves the slides, this page and the video.
%
% Every number comes from the abstract and tables of ../main.tex. Nothing here
% reports the later thesis experiments.
%
% Grid: margin 12, gutter 7, panel widths 96 / 88 / 98 at x = 12 / 115 / 210.
% Columns are deliberately unequal: the result is why anyone stops, the problem
% is one sentence of setup. Inset is 8 mm on every side of every panel, and
% panel heads sit on that same 8 mm so they align with the text they govern.
%
% Type scale 46 / 26 / 20 / 15 / 11 / 9. Steps are 1.2x or wider; anything
% closer stops reading as hierarchy and starts reading as a mistake.
%
% Colour: one dark ink for all three panel heads, with a 3 mm accent tab, so
% the accents stay reserved. Vermillion is the Severe class and nothing else,
% blue and orange are the two branches and nothing else, green is improvement
% and nothing else. Palette is Okabe-Ito, readable with colour vision
% deficiency, and no meaning is carried by colour alone.
%
% Build:
%   pdflatex -interaction=nonstopmode infographic.tex
%   pdftoppm -png -scale-to-x 1920 -scale-to-y 1080 infographic.pdf infographic

% One card: white ground, hairline, head bar clipped inside the rounding so all
% four corners share a radius.
\newcommand{\card}[4]{%
  \begin{scope}
    \clip[rounded corners=2mm] (#1,26) rectangle ++(#2,122);
    \fill[white] (#1,26) rectangle ++(#2,122);
    \fill[inkDark] (#1,136) rectangle ++(#2,12);
    \fill[#4] (#1,136) rectangle ++(3,12);
  \end{scope}
  \draw[cardLine, line width=0.25mm, rounded corners=2mm] (#1,26) rectangle ++(#2,122);
  \node[anchor=west, text=white, font=\fontsize{15}{17}\selectfont\bfseries]
       at (#1+8,141.6) {\lbl{#3}};}
