\subsection{Mục tiêu và phạm vi}
\begin{frame}
\frametitle{Bài toán và mục tiêu}
\textbf{Bối cảnh:} hệ thống hỗ trợ chẩn đoán đau thắt lưng gồm hai mô-đun (phân vùng và đánh giá); giai đoạn này tập trung mô-đun đánh giá (grading) với hai bài toán cốt lõi:
\begin{itemize}\small
    \item[\textbf{1.}] \textbf{Mất cân bằng lớp:} lớp Severe quá hiếm ($\sim$4\%) nên dễ bị bỏ sót.
    \item[\textbf{2.}] \textbf{Mở rộng nhãn:} thêm nhãn bệnh mới mà không huấn luyện lại; head phân loại cố định không làm được.
\end{itemize}
\vspace{0.3em}
\textbf{Mục tiêu:}
\begin{itemize}\small
    \item[--] Xây pipeline 2 giai đoạn và khảo sát, chọn mô hình nền cho từng giai đoạn.
    \item[--] Tinh chỉnh grading để giải quyết hai bài toán trên (cải thiện Severe + mở nhãn zero-shot).
    \item[--] Kiểm chứng tổng quát hóa sang dữ liệu khác.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Phạm vi đề tài}
\begin{itemize}
    \item[--] \textbf{Dữ liệu:} chỉ ảnh MRI cột sống thắt lưng (không xử lý X-quang/CT), chủ yếu mặt cắt Sagittal T2/STIR; chưa khai thác mặt cắt Axial.
    \item[--] \textbf{Cấu trúc giải phẫu:} đốt sống, đĩa đệm, ống sống và tủy sống.
    \item[--] \textbf{Bệnh lý đánh giá:} hẹp ống sống, hẹp lỗ liên hợp trái/phải (RSNA); mở rộng sang Pfirrmann, Modic, thoát vị, phồng đĩa đệm, trượt đốt sống (SPIDER).
\end{itemize}
\pause
\textbf{Phạm vi giai đoạn thực tập:}
\begin{itemize}
    \item[--] Tiền đề cho Đồ án tốt nghiệp: tập trung chọn mô hình + tinh chỉnh grading.
    \item[$\rightarrow$] Pipeline hoàn chỉnh + phần mềm giao diện: để dành cho Đồ án tốt nghiệp.
\end{itemize}
\end{frame}

\subsection{Bộ dữ liệu}
\begin{frame}
\frametitle{Các bộ dữ liệu sử dụng}
\textbf{Đánh giá bất thường}
\begin{itemize}\small
    \item[--] \textbf{RSNA 2024}~\cite{rsna2024dataset}: MRI cột sống thắt lưng, phân mức Normal/Mild/Moderate/Severe cho hẹp ống sống và hẹp lỗ liên hợp trái/phải. Tập chính để huấn luyện và đánh giá.
    \item[--] \textbf{SPIDER}~\cite{vansluijs2022spider}: 8 nhãn bệnh (Pfirrmann, thoát vị, phồng đĩa đệm, trượt đốt sống, Modic\ldots). Dùng kiểm chứng tổng quát hóa sang dữ liệu khác.

\textbf{Phân vùng cấu trúc}
\begin{itemize}\small

\subsection{Động lực thiết kế}
\begin{frame}
\frametitle{Động lực thiết kế}
\textbf{Vấn đề:} Các mô hình đánh giá hiện nay vẫn bỏ sót nhiều ca nặng (Severe) do lớp này quá hiếm ($\sim$4\%) và vùng tổn thương rất nhỏ trong khối ảnh; ví dụ SpineNetV2 chỉ đạt Recall 11.5\% trên lớp Severe.
\begin{itemize}
    \item[--] \textbf{Cần một nhánh chú ý:} tập trung đúng kênh đặc trưng và vùng không gian của tổn thương nhỏ, thay vì xử lý đều toàn ảnh.
    \item[--] \textbf{Cần một nhánh tri thức nền:} (a) bù dữ liệu Severe quá ít bằng đặc trưng tổng quát học sẵn; (b) đặt ảnh--văn bản chung không gian để \textbf{mở nhãn mới bằng prompt} (zero-shot).
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Phụ lục: các công thức cốt lõi}
\textbf{CBAM:}
\begin{align}
M_c(F) &= \sigma\!\big(\text{MLP}(\text{AvgPool}(F)) + \text{MLP}(\text{MaxPool}(F))\big), \notag\\
M_s(F') &= \sigma\!\big(f^{7\times7\times7}([\text{AvgPool}_c(F');\,\text{MaxPool}_c(F')])\big), \notag\\
F'' &= M_s(M_c(F)\otimes F)\otimes(M_c(F)\otimes F). \notag
\end{align}
\textbf{Focal Loss (có trọng số lớp, $\gamma=2$):}
\begin{equation}
\mathrm{FL}_t(p_t) = -\,w_{c(t)}\,(1-p_t)^{\gamma}\,\log(p_t). \notag
\end{equation}
\textbf{Dự đoán zero-shot bằng đối sánh cosine với prompt $\{t_k\}$:}
\begin{equation}
\hat{y} = \arg\max_{k}\; s\cdot \hat{f}_{\text{img}}^{\top} t_k. \notag
\end{equation}
\end{frame}
